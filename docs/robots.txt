#      أمر دليل العميل في tools/build-guides.sh ثم أعد تشغيل الملف
#      (وسم noindex في الدليل يولّده المولِّد، فحذفه باليد يعود عند التوليد)
#   ٣. اضبط TRIAL_MODE = false في docs/index.html
